4. Python Dialer Integration Snippet 
import requests

BASE_URL = "https://ais-pre-uyi5mhe62n4gltsgek6tso-893275087870.asia-southeast1.run.app"

# Start Outbound Call
session = requests.post(f"{BASE_URL}/api/dialer/call/start", json={
    "direction": "outbound",
    "customer_name": "Michael Vance",
    "campaign": "Payment Reminder"
}).json()

print(f"Call ID: {session['call_id']}")
print(f"Greeting Audio: {BASE_URL}{session['audio_url']}")

# Send User Audio/Text and Check for Transfer
reply = requests.post(f"{BASE_URL}/api/dialer/call/interact", json={
    "call_id": session['call_id'],
    "text": "Can you transfer me to an agent?"
}).json()

if reply["action"] == "transfer":
    target_ext = reply["transfer"]["extension"]
    print(f"Bridge call to SIP Extension: {target_ext} ({reply['transfer']['sip_uri']})")
